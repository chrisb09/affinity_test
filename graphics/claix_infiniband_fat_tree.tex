% CLAIX-2023 InfiniBand fat-tree topology (schematic).
%
% Two-tier fat-tree derived from `scontrol show topology`:
%   * Level 1: the spine group `spines1234` aggregates every leaf switch; the
%     name indicates four physical spine switches.
%   * Level 0: leaf switches `ibsw00`..`ibsw10`.
%       - `ibsw00`..`ibsw08` carry the CPU-only nodes (c23ms/c23mm/c23ml).
%       - `ibsw09`/`ibsw10` carry the accelerated nodes (c23g).
%   * CPU and GPU nodes attach to different leaf switches, so CPU<->GPU traffic
%     always crosses the spine (3 switch hops), while two nodes on the same leaf
%     communicate in a single switch hop.
%   * Node boxes are intentionally left anonymous (the node-id to switch mapping
%     is not a contiguous block); the "CPU nodes"/"GPU nodes" labels identify the
%     two node classes. Every drawn leaf shows its first and last child plus an
%     ellipsis.
%
% Connector convention: a child attaches at its top edge; the parent's end fans
% across its bottom edge instead of meeting at a single centre point. Using
% compass bearings (180 = down), a parent with n children spreads its anchors
% linearly over 135..225 degrees, i.e. 225/180/135 for three children. In TikZ
% angle terms (0 = east, counter-clockwise) that range is 315..225, so three
% anchors are 315 (right), 270 (down) and 225 (left).
%
% Intended to be \input inside a figure or a standalone document.
\begin{tikzpicture}[
  x=1mm, y=1mm, font=\sffamily\fontsize{8}{9.5}\selectfont,
  >=Stealth,
  spine/.style={rounded corners=2.5pt, draw=black!70, line width=0.9pt, fill=black!6,
    minimum width=140mm, minimum height=10mm, align=center, inner sep=1.5pt},
  leafcpu/.style={rounded corners=2.5pt, draw=blue!55, line width=0.9pt, fill=blue!6,
    minimum width=17mm, minimum height=8mm, align=center, inner sep=1pt},
  leafgpu/.style={rounded corners=2.5pt, draw=orange!85, line width=0.9pt, fill=orange!10,
    minimum width=17mm, minimum height=8mm, align=center, inner sep=1pt},
  nodecpu/.style={rounded corners=2pt, draw=blue!55, line width=0.8pt, fill=blue!3,
    minimum width=15mm, minimum height=8mm, inner sep=1pt},
  nodegpu/.style={rounded corners=2pt, draw=orange!85, line width=0.8pt, fill=orange!8,
    minimum width=15mm, minimum height=8mm, inner sep=1pt},
  link/.style={draw=black!55, line width=0.6pt},
  tree/.style={draw=black!45, line width=0.6pt},
  path3/.style={draw=red!75!black, line width=1.5pt},
  path1/.style={draw=green!45!black, line width=1.5pt},
  tierlabel/.style={font=\sffamily\fontsize{8}{9}\selectfont, text=black!60,
    anchor=west},
  annot/.style={font=\sffamily\fontsize{8}{9.5}\selectfont, align=left},
  ell/.style={font=\sffamily\fontsize{11}{11}\selectfont, text=black!55},
  hopmark/.style={circle, draw=red!75!black, fill=white, inner sep=0.5pt,
    font=\sffamily\fontsize{6}{6.5}\selectfont, text=red!75!black},
  hopmark1/.style={circle, draw=green!45!black, fill=white, inner sep=0.5pt,
    font=\sffamily\fontsize{6}{6.5}\selectfont, text=green!45!black}
]

% ---- Level labels ---------------------------------------------------------
\node[tierlabel] at (0,84) {Level 1};

\node[spine] (spine) at (90,84) {\textbf{spine switches 1--4}};

\node[leafcpu] (l0)  at (26,69)  {\texttt{ibsw00}};
\node[leafcpu] (l8)  at (72,69)  {\texttt{ibsw08}};
\node[leafgpu] (l9)  at (112,69) {\texttt{ibsw09}};
\node[leafgpu] (l10) at (154,69) {\texttt{ibsw10}};
\node[ell] at (49,69) {$\cdots$};
